// =============================================================================
// BIOS — Business Intelligence on Sustainability
// Modelo de dados (MongoDB + Prisma ORM v6)
// =============================================================================
//
// Convenções:
//  - Chave primária de cada coleção no formato `id_<entidade>` (ObjectId, mapeado
//    para `_id` no Prisma via @map("_id") @db.ObjectId);
//  - Chave estrangeira no formato `<entidade>_id`; empresas são referenciadas
//    pela chave natural `cnpj` (14 dígitos, sem máscara);
//  - Campos em snake_case; coleções em snake_case no plural, exceto as três
//    coleções das camadas de dados (`evidence_raw`, `evidence_normalized`,
//    `company_profile`), que mantêm o nome adotado na arquitetura;
//  - Auditoria temporal: `created_at`, `updated_at` e, quando há remoção lógica,
//    `deleted_at`;
//  - Índices nomeados no padrão `idx_<coleção>_<campos>`;
//  - Tabelas marcadas como "subdocumento embutido" não são coleções próprias:
//    são tipos compostos (Prisma `type`) gravados dentro de outro documento;
//  - Relações são lógicas: o MongoDB não impõe integridade referencial, que
//    fica a cargo da camada de aplicação (repositórios).
//
// Segurança de dados:
//  - E-mail e endereço de usuário cifrados com AES-256 ([NFSE03]); a senha
//    recebe hash bcrypt, cifrado em seguida com AES-256;
//  - Chaves de cifragem NUNCA são armazenadas no banco: ficam em variáveis de
//    ambiente/cofre de segredos. O banco guarda apenas a versão da chave usada
//    (`encryption_key_version`), permitindo rotação gradual;
//  - Tokens de uso único, sessões e chaves de API são armazenados apenas como
//    hash (SHA-256 — entradas de alta entropia, sem necessidade de bcrypt);
//  - Nenhuma coleção armazena dado pessoal de sócio, administrador ou
//    colaborador das empresas consultadas ([INL02]).
//
// Retenção:
//  - Coleções de log e tokens recebem índice TTL do MongoDB (criado via
//    script de migração, pois o Prisma não declara índices TTL).
// =============================================================================

Project bios {
  database_type: 'MongoDB'
  Note: 'BIOS — Business Intelligence on Sustainability. Consolida evidências ambientais públicas por CNPJ e calcula, para os ODS 13 e 15, índice de aderência e nível de confiança explicáveis.'

Table users {
  id_user objectid [primary key]
  user_name varchar
  email_encrypted varchar [note: 'E-mail cifrado com AES-256']
  email_hash varchar [note: 'Índice cego: hash determinístico do e-mail normalizado, usado no login e na verificação de unicidade']
  password_encrypted varchar [note: 'Hash bcrypt da senha, cifrado com AES-256']
  address_encrypted varchar [null, note: 'Endereço completo serializado e cifrado com AES-256']
  encryption_key_version int [default: 1, note: 'Versão da chave AES usada nos campos cifrados deste documento']
  role user_role [default: 'USER']
  is_active boolean [default: false, note: 'Falso até a confirmação do e-mail (RF28)']
  email_confirmed_at timestamp [null]
  last_login_at timestamp [null]
  password_changed_at timestamp [null]
  created_at timestamp
  updated_at timestamp

  Note: 'Contas de usuário. Sem soft delete: a exclusão de conta (RF27) remove o documento e seus dependentes, atendendo ao direito de eliminação da LGPD; tentativas de login associadas têm `user_id` anulado.'

  Indexes {
    email_hash [unique, name: 'idx_users_email_hash']
    role [name: 'idx_users_role']
  }
}

Table user_tokens {
  id_user_token objectid [primary key]
  user_id objectid [ref: > users.id_user]
  purpose user_token_purpose
  token_hash varchar [note: 'Hash SHA-256 do token enviado por e-mail; o valor puro nunca é persistido']
  pending_email_encrypted varchar [null, note: 'Novo e-mail cifrado, presente apenas quando purpose = EMAIL_CHANGE']
  expires_at timestamp
  used_at timestamp [null]
  created_at timestamp
  updated_at timestamp

  Note: 'Tokens de uso único e validade limitada enviados por e-mail (confirmação de conta, troca de e-mail e recuperação de senha). TTL: removidos 7 dias após expires_at.'

  Indexes {
    token_hash [unique, name: 'idx_user_tokens_token_hash']
    (user_id, purpose, used_at) [name: 'idx_user_tokens_user_purpose_used']
    expires_at [name: 'idx_user_tokens_expires_at']
  }
}

Table user_sessions {
  id_user_session objectid [primary key]
  user_id objectid [ref: > users.id_user]
  jti_hash varchar [note: 'Hash do identificador único (jti) do JWT emitido']
  ip varchar
  user_agent varchar [null]
  expires_at timestamp
  revoked_at timestamp [null]
  revocation_reason session_revocation_reason [null]
  created_at timestamp

  Note: 'Sessões JWT revogáveis (RF23, RF24, NFSE05). Um token só é aceito se a sessão existir, não estiver revogada e não estiver expirada. TTL: removidas 30 dias após expires_at.'

  Indexes {
    jti_hash [unique, name: 'idx_user_sessions_jti_hash']
    (user_id, revoked_at) [name: 'idx_user_sessions_user_revoked']
    expires_at [name: 'idx_user_sessions_expires_at']
  }
}

Table api_keys {
  id_api_key objectid [primary key]
  user_id objectid [ref: > users.id_user]
  key_name varchar [note: 'Nome de identificação escolhido pelo usuário']
  key_prefix varchar [note: 'Prefixo público da chave, exibido na listagem e usado na busca']
  key_hash varchar [note: 'Hash SHA-256 da chave completa; a chave só é exibida uma vez, na geração (NFSE01)']
  last_used_at timestamp [null]
  revoked_at timestamp [null]
  created_at timestamp
  updated_at timestamp

  Note: 'Chaves de API para consumo programático (RF16, RF17). Revogadas em massa na exclusão da conta.'

  Indexes {
    key_prefix [unique, name: 'idx_api_keys_key_prefix']
    key_hash [unique, name: 'idx_api_keys_key_hash']
    (user_id, revoked_at) [name: 'idx_api_keys_user_revoked']
  }
}

  id_data_source objectid [primary key]
  slug varchar [note: 'Identificador estável usado pelos adapters (ex.: BRASIL_API, OPEN_CNPJ, MINHA_RECEITA, IBGE_CNAE, IBAMA, CGU_CEIS, CGU_CNEP, GHG_PROTOCOL)']

  rate_limit_per_minute int [null, note: 'Limite respeitado pelo adapter (NFPD01)']
  cache_ttl_seconds int [null, note: 'Janela de validade do cache das consultas a esta fonte (NFDE01)']
  request_timeout_ms int [null]
  is_enabled boolean [default: true]
  created_at timestamp
  updated_at timestamp

  Note: 'Fontes externas de dados, populadas via seed. Mantê-las como entidade permite reordenar a cadeia de fallback, desabilitar uma fonte ou ajustar limites sem novo deploy.'

  Indexes {
    slug [unique, name: 'idx_data_sources_slug']
    (role, is_enabled, fallback_priority) [name: 'idx_data_sources_role_enabled_priority']
  }

  contributions score_contribution[] [null]
  rules_version varchar [note: 'Versão das regras de pontuação usadas no cálculo, para rastrear recálculos']
  uf varchar [note: 'Desnormalizado de company_profile.address.uf']
  city varchar [note: 'Desnormalizado de company_profile.address.city']

  }
}

  id_sector_score_snapshot objectid [primary key]
  ods_id objectid [ref: > ods.id_ods]
  cnae_division varchar [note: 'Divisão CNAE (2 dígitos), granularidade de setor adotada para evitar amostras pequenas']
  window_start timestamp
  window_end timestamp
  average_score double [null]
  average_confidence double
  company_count int [note: 'Empresas do setor com documento em company_ods_scores na janela']
  scored_company_count int [note: 'Empresas com índice não nulo na janela']
  rules_version varchar
  created_at timestamp

  Note: 'Agregados setoriais por janela de tempo (RF30, indicador IA05). Armazena apenas médias por setor, nunca série histórica por empresa individual.'

  Indexes {
    (ods_id, cnae_division, window_start) [unique, name: 'idx_sector_score_snapshots_ods_division_window']
  }
}
